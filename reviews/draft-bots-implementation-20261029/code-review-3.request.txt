def validate_config(config, ids, plans=None):
    if not isinstance(config, dict) or config.get('schemaVersion') != 1:
        raise ValueError('botPolicy: schemaVersion non supportata')
    if config.get('match', 'BO3') not in ('BO1', 'BO3', 'unknown'):
        raise ValueError('botPolicy: contesto match non supportato')
    if not isinstance(config.get('cards', {}), dict) or not set(config.get('cards', {})) <= set(ids):
        raise ValueError('botPolicy: carta sconosciuta')
    relations = config.get('relations', [])
    if not isinstance(relations, list) or any(not isinstance(r, dict) for r in relations) or len({r.get('id') for r in relations}) != len(relations):
        raise ValueError('botPolicy: relazioni duplicate o non valide')
    for relation in relations:
        if not isinstance(relation.get('id'), str) or not relation['id']:
            raise ValueError('botPolicy: identità relazione mancante')
        if any(field not in relation for field in ('referenceSupport', 'saturation', 'strength')):
            raise ValueError('botPolicy: riferimento della relazione mancante')
        for field in ('referenceSupport', 'referencePayoffs', 'saturation', 'strength', 'threshold', 'resourceBudget'):
            value = relation.get(field, 0)
            if type(value) not in (int, float) or not 0 <= value <= 100:
                raise ValueError('botPolicy: parametro relazione non valido')
        if relation.get('saturation', 8) < relation.get('referenceSupport', 0):
            raise ValueError('botPolicy: saturazione inferiore al riferimento')
    plan_coverage = config.get('planCoverage', {})
    if not isinstance(plan_coverage, dict) or plans is not None and not set(plan_coverage) <= set(plans):
        raise ValueError('botPolicy: piano sconosciuto')
    for state in plan_coverage.values():
        if not isinstance(state, dict) or state.get('status') not in ('supported', 'not_applicable', 'unsupported') or not isinstance(state.get('provenance'), dict) or not isinstance(state.get('relations', []), list) or not set(state.get('relations', [])) <= {r['id'] for r in relations}:
            raise ValueError('botPolicy: copertura del piano non valida')
    for profile in config.get('cards', {}).values():
        if not isinstance(profile, dict) or not isinstance(profile.get('provenance'), dict):
            raise ValueError('botPolicy: provenienza profilo mancante')

            raise ValueError('botPolicy: simboli non validi')

            raise ValueError('botPolicy: costo variabile non valido')
        for key in ('mana', 'curve', 'reference', 'synergy'):
            if key in profile and not isinstance(profile[key], dict):
                raise ValueError('botPolicy: descrittore non valido: ' + key)
        reference = profile.get('reference', {})

            raise ValueError('botPolicy: contesto di riferimento mancante')
        for key in ('support', 'payoffs'):
            values = reference.get(key, {})

def payload(cards, active, event, config, table_ready, plans=None):
    profiles = {}
    for card in cards:
        if card['id'] not in active:
            continue
        profile = describe(card)
        override = config.get('cards', {}).get(card['id'], {})
        for key, value in override.items():
            if key in ('mana', 'curve', 'synergy', 'reference'):
                profile[key] = {**profile[key], **value}
            else:
                profile[key] = value
        profiles[card['id']] = profile
    active_cards = [c for c in cards if c['id'] in active]
    minimum = bool(active_cards) and all(c.get('basic') or (
        type(c.get('score')) in (int, float) and 0 <= c['score'] <= 5
        and profiles[c['id']]['status'] == 'supported' and profiles[c['id']].get('roles')
        and isinstance(profiles[c['id']].get('mana'), dict)) for c in active_cards)
    relations = config.get('relations', [])
    plan_coverage = config.get('planCoverage', {})
    covered = bool(relations) and bool(plans) and all(plan_coverage.get(plan, {}).get('status') in ('supported', 'not_applicable') for plan in plans) and all(
        p.get('synergy', {}).get('status') in ('supported', 'not_applicable')
        and p.get('reference', {}).get('status') == 'supported' for p in profiles.values())
    semantic = {'version': VERSION, 'profiles': {
        key: {k: ({key: v[key] for key in ('status', 'support', 'payoffs') if key in v} if k == 'reference' else v) for k, v in value.items() if k != 'provenance'} for key, value in profiles.items()},
        'relations': [{k: r[k] for k in ('id', 'referenceSupport', 'referencePayoffs', 'saturation', 'strength', 'threshold', 'resource', 'resourceBudget') if k in r} for r in relations], 'match': config.get('match', 'BO3'),
        'plans': {key: {'status': value['status'], 'relations': value.get('relations', [])} for key, value in plan_coverage.items()},
        'minimum': event['minDeckSize'], 'extraBasics': event.get('extraBasicLands', False),
        'maxCopies': event.get('maxCopies')}
    revision = hashlib.sha256(json.dumps(semantic, sort_keys=True, ensure_ascii=True).encode()).hexdigest()
    missing = []
    if not table_ready:

    return {'schemaVersion': 1, 'inferenceVersion': VERSION, 'revision': revision, 'algorithmRevision': algorithm_revision,
            'available': bool(table_ready and minimum), 'status': 'experimental',
            'missing': missing, 'gaps': gaps, 'coverage': {
                'active': len(active_cards), 'minimum': sum(bool(p.get('roles')) and p['status'] == 'supported' for p in profiles.values()),
                'synergies': covered,'plans': {plan: plan_coverage.get(plan, {'status':'unsupported'}) for plan in plans or []}}, 'profiles': profiles, 'relations': relations,
            'match': config.get('match', 'BO3'), 'minDeckSize': event['minDeckSize'],

def read_json(path):
    def invalid(value):raise ValueError('Numero JSON non valido: '+value)
    return json.loads(path.read_text(encoding='utf-8-sig'),parse_constant=invalid)


def confined(base, relative):
    path = (base / relative).resolve()
    if not path.is_relative_to(base.resolve()):
        raise ValueError(f'Percorso esterno alla cartella: {relative}')
    return path


def normalize_card(source, document):
    if not isinstance(source,dict):raise ValueError(document+': carta non valida')
    card=dict(source)
    if document=='printings.json':card.setdefault('url','')
    identity=str(card.get('id','senza id'))
    def fail(field):raise ValueError(f'{document}/{identity}: campo carta mancante o non valido: {field}')
    for field in ('id','name','set','type','rarity','image','url'):
        if not isinstance(card.get(field),str):fail(field)
    if not card['id'] or not card['name']:fail('id / name')
    if type(card.get('number')) not in (int,str):fail('number')
    for field in ('colors','identity'):
        if not isinstance(card.get(field),list) or any(c not in list('WUBRG') for c in card[field]):fail(field)
    for canonical,alias,types in (('cost','mana',(str,)),('mv','cmc',(int,float)),('oracle','text',(str,))):
        value=card.get(canonical,card.get(alias))
        if type(value) not in types:fail(canonical+' / '+alias)
        card[canonical]=value;card[alias]=value
    card.setdefault('full',card['name']);card.setdefault('large',card['image'])
    card.setdefault('faces',[]);card.setdefault('keywords',[]);card.setdefault('prints',[]);card.setdefault('legalities',{})
    main_type=card['type'].split(' // ')[0]
    for field,default in (('basic',bool(re.search(r'\bBasic\b',main_type) and re.search(r'\bLand\b',main_type))),('land',bool(re.search(r'\bLand\b',main_type))),('creature',bool(re.search(r'\bCreature\b',main_type)))):
        card.setdefault(field,default)
        if type(card[field]) is not bool:fail(field)
    if not isinstance(card['full'],str):fail('full')
    if not isinstance(card['faces'],list) or any(not isinstance(f,dict) or not isinstance(f.get('name'),str) for f in card['faces']):fail('faces')
    card['faces']=[{'mana_cost':'','oracle_text':'','type_line':'','power':'','toughness':'',**face} for face in card['faces']]
    if not isinstance(card['keywords'],list) or any(not isinstance(k,str) for k in card['keywords']):fail('keywords')
    if not isinstance(card['prints'],list) or not isinstance(card['legalities'],dict):fail('prints / legalities')
    if any(not isinstance(p,dict) or not isinstance(p.get('finishes',[]),list) for p in card['prints']):fail('prints.finishes')
    card['prints']=[{'number':'','image':'','finishes':[],**printing} for printing in card['prints']]
    return card


class Expansion:
    def __init__(self, manifest_path):
        self.path = Path(manifest_path)
        self.directory = self.path.parent
        self.manifest = read_json(self.path)
        m = self.manifest
        if m.get('schemaVersion') != 1:
            raise ValueError(f'{self.path}: schemaVersion non supportata')
        if not re.fullmatch(r'[a-z0-9-]+', m.get('id','')):
            raise ValueError('Identificatore espansione non valido')
        for field in ('name','code','date','description','datasets','presentation','storage','features'):
            if not m.get(field): raise ValueError(f'{m["id"]}: manca {field}')
        for field in ('name','code','date','description'):
            if not isinstance(m[field],str):raise ValueError(f'{m["id"]}: {field} deve essere una stringa')
        self.data = {key: read_json(confined(self.directory, m['datasets'][key]))
                     for key in DATASETS if key in m['datasets']}
        missing = set(DATASETS) - self.data.keys()
        if missing: raise ValueError(f'{m["id"]}: documenti mancanti: {sorted(missing)}')
        for key in ('products','limited'):
            if self.data[key].get('schemaVersion')!=1:raise ValueError(f'{m["id"]}: versione {key} non supportata')
        for key in ('cards','printings'):
            if not isinstance(self.data[key].get('cards'),list):raise ValueError(f'{m["id"]}/{key}.json: manca cards')
            self.data[key]['cards']=[normalize_card(card,key+'.json') for card in self.data[key]['cards']]
        if type(self.data['cards'].get('complete',False)) is not bool:raise ValueError(m['id']+'/cards.json: complete deve essere booleano')
        self.data['cards'].setdefault('print_count',len(self.data['printings']['cards']) or sum(max(1,len(card['prints'])) for card in self.data['cards']['cards']))
        self.validate()
        self.availability = self.capabilities()

    @property

        """Supply shared-page pool keys without renaming existing saved records."""
        storage=dict(self.manifest['storage'])
        for kind in ('draft','sealed'):
            storage.setdefault(kind+'Pool',self.manifest['id']+'-'+kind+'-pool-v1')
        storage.setdefault('sealedSimulator',self.manifest['id']+'-sealed-simulator-v1')
        return storage

    @property
    def cards(self):
        ratings = self.data['ratings'].get('cards',{})
        sentiments = self.data['sentiment'].get('cards',{})
        result=[]
        for source in self.data['cards']['cards']:
            card={'score':None,'role':'','why':'','condition':'','evidence':[],'community':None,
                  'forum_note':'','forum_link':'','build':False,**source, **ratings.get(source['id'],{})}
            if source['id'] in sentiments: card['sentiment']=sentiments[source['id']]
            card['keywords']=list(source.get('keywords',[]))
            for alias in self.data['ui'].get('keywordAliases',[]):
                if re.search(alias['pattern'],card['oracle'],re.I) and alias['keyword'] not in card['keywords']:card['keywords'].append(alias['keyword'])
            found=[combo for combo in self.data['combos'] if card['id'] in combo['cards']]
            families={}
            for combo in found:
                q=.65*combo['practical']+.35*combo['power']
                families[combo['family']]=max(q,families.get(combo['family'],0))
            scores=sorted(families.values(),reverse=True)
            score=min(5,scores[0]+min(.5,.2*sum(max(0,x-2) for x in scores[1:4]))) if scores else None
            card.update(combos=[c['id'] for c in found],comboFamilies=len(families),comboScore=int(score*2+.5)/2 if score is not None else None)
            result.append(card)
        return result

    def default_event_id(self, kind):
        """Prefer a usable event of this kind, keeping historical IDs when possible."""
        events=self.data['limited']['events']
        candidates=[key for key,event in events.items() if event['kind']==kind]
        if kind in candidates:candidates.remove(kind);candidates.insert(0,kind)
        return next((key for key in candidates if self.event_availability(key)['available']),candidates[0] if candidates else None)

    def resolve_event_id(self, event_id):
        if event_id not in self.data['limited']['events'] and event_id in ('draft','sealed'):
            return self.default_event_id(event_id)
        return event_id

    def product_for(self,event_id):
        event=self.data['limited']['events'].get(event_id)
        return self.data['products']['products'].get(event['product'],{}) if event else {}

    def product_printings(self,event_id):
        recipe=self.product_for(event_id).get('collation') or {}
        ids={id_ for rule in recipe.get('recipe',[]) for branch in rule['branches'] for id_ in recipe['groups'][branch['group']]}
        for a,b in recipe.get('partners',[]):
            if a in ids or b in ids:ids.update((a,b))
        return ids

    def event_availability(self,event_id):
        event=self.data['limited']['events'].get(event_id)

        if not self.data['cards'].get('complete'):missing.append(REASONS['completeCards'])

    def draft_format(self, event_id='draft'):
        event_id=self.resolve_event_id(event_id)
        event = self.data['limited']['events'][event_id]
        product = self.data['products']['products'][event['product']]
        archetypes = self.data['archetypes']
        catalog = self.printing_catalog()
        cards=[]
        for printing in self.data['printings']['cards']:
            card = {**printing}
            review = catalog.get(card['id']) or {}
            card.update(score=review.get('score'),role=review.get('role',''))
            card.update(power=review.get('power'),toughness=review.get('toughness'),faces=review.get('faces',[]))
            card['catalogId'] = review.get('id')
            card['archetypes'] = [k for k,a in archetypes.items() if card['name'] in a['core']+a['support']]
            cards.append(card)
        active=self.product_printings(event_id)
        engine_fields=('id','full','colors','identity','cost','mv','rarity','basic','land','creature','score','role','oracle','archetypes')
        spec=product.get('collation') or {}
        engine={'engineVersion':2,'event':{key:event.get(key) for key in ('id','seats','packsPerPlayer','picksPerTurn','directions')},
                'packSize':product['packSize'],
                'collation':{key:spec.get(key) for key in ('schemaVersion','groups','partners','recipe')},
                'cards':[{key:card.get(key) for key in engine_fields} for card in cards if card['id'] in active],
                'archetypeOrder':list(archetypes)}
        revision = hashlib.sha256(json.dumps(engine,sort_keys=True,ensure_ascii=True).encode()).hexdigest()
        compatibility=self.manifest.get('draftReplayCompatibility',{}).get(event_id,{})
        previous=compatibility.get('previousRevisions',[]) if compatibility.get('revision')==revision else []
        from bot_profiles import payload
        bot_policy=payload(cards, active, event, event.get('botPolicy', {}),
                           event.get('kind')=='draft' and self.draft_event_availability(event_id)['available'], archetypes)
        return {'version':1,'id':event['id'],'set':self.manifest['code'],'name':event['name'],
                'botPolicy':bot_policy,'minDeckSize':event['minDeckSize'],
                'extraBasicLands':event.get('extraBasicLands',False),'maxCopies':event.get('maxCopies'),
                'seats':event['seats'],'rounds':event['packsPerPlayer'],'packSize':product['packSize'],'productName':product['name'],
                'picksPerTurn':event['picksPerTurn'],'directions':event.get('directions',[]),
                'assumptions':product.get('assumptions',[]),'cards':cards,'archetypes':archetypes,
                'collation':product.get('collation'),'revision':revision,
                'compatibleRevisions':previous,
                'legacyReplay':event.get('legacyReplay',False) and self.manifest.get('legacyDraftRevision') in previous, 'ui':self.data['ui']}

    def runtime(self):
        return {'schemaVersion':1,'id':self.manifest['id'],'name':self.manifest['name'],'code':self.manifest['code'],
                'storage':self.storage,'events':self.data['limited']['events'],
                'routes':{key:p['url'] for key,p in self.manifest['presentation']['pages'].items()},
                'availability':self.availability,'ui':self.data['ui'],
                'defaultEvents':{kind:self.default_event_id(kind) for kind in ('draft','sealed')},
                'eventAvailability':{key:self.event_availability(key) for key in self.data['limited']['events']}}


def load_expansions():
    expansions = [Expansion(confined(ROOT,entry['manifest'])) for entry in read_json(ROOT/'expansions.json')]
    if len({e.manifest['id'] for e in expansions}) != len(expansions): raise ValueError('Espansioni duplicate')
    storage={}
    for expansion in expansions:
        spec=expansion.storage
        keys=[v for k,v in spec.items() if k!='poolPrefix']
        if 'poolPrefix' in spec:
            keys.extend(spec['poolPrefix']+mode+'-v1' for mode in expansion.data['limited']['events'])
        if 'draftTable' in spec:
            keys.extend(spec['draftTable']+'-'+mode for mode,event in expansion.data['limited']['events'].items() if event['kind']=='draft' and mode!='draft')
        for key in keys:
            if key in storage and storage[key]!=expansion.manifest['id']:raise ValueError(f'Chiave di salvataggio condivisa da espansioni diverse: {key}')
            storage[key]=expansion.manifest['id']
    return expansions

### Dati e disponibilità: decisioni progettuali

La qualità di base usa `ratings.json`, con scala e provenienza Limited del progetto. GIH e altre misure osservate restano evidenze separate per rivedere manualmente i voti discordanti, considerando contesto e campione; nessuna sostituzione automatica o media nascosta. Un voto mancante non diventa 2,5.

Prima di applicare scarti contestuali, documentare quale situazione assume il voto della scala usata e le eventuali eccezioni per carta: per esempio carta nel suo mazzo tipico con sostegno ordinario, oppure valore autonomo. Non attribuire retroattivamente un significato uniforme ai voti esistenti se le fonti non lo sostengono. I contributi sono scarti, anche negativi, da quel riferimento, non bonus per benefici già compresi nel voto. Se il riferimento manca è una lacuna da risolvere o disattivare secondo le soglie, non una definizione da inventare. Tre fixture controllano sostegno di riferimento, assente ed eccezionale, senza cambiare il voto originale per farle passare.

Prerequisiti per rendere disponibile la nuova versione: tavolo realmente disponibile e, sul 100% delle stampe attive non base, voto valido e profilo minimo esplicito di costo/mana e ruoli. Le soglie valgono per tutti i profili della nuova versione che usano capacità condivise, non soltanto Adattivo. Per ogni capacità aggiuntiva, tutte le carte attive devono avere uno stato esplicito: descrizione supportata, non pertinente verificato oppure non supportato. Un semplice campo vuoto non vale come non pertinente. Le annotazioni e le inferenze Oracle hanno provenienza, revisione e controlli mirati.

Una capacità è dichiarata coperta solo se nessuna carta pertinente resta non supportata e le relative fixture passano. Se mancano i dati minimi, la nuova versione non è disponibile nell'evento: restano soltanto le politiche storiche già disponibili, senza abilitare nuovi tavoli. Se i dati minimi sono completi ma una capacità aggiuntiva è parziale, tutti i profili aggiornati sono sperimentali con lo stesso elenco di lacune e gli stessi contributi non sostenuti disattivati, con valutazione conservativa e spiegata. Potenza pura e Archetipo forzato non aggirano soglie relative a componenti che usano. Nessuna etichetta di competenza generale finché una meccanica del piano valutato è ignorata. Non occorre codificare ogni possibile interazione: il manifest delle capacità dichiara quelle modellate e il benchmark circoscrive le conclusioni.

La granularità delle capacità è fissata prima di annotazione e taratura. In particolare la componente di sinergie contestuali comprende l'insieme delle relazioni previste per i piani dell'evento, non soltanto il singolo asse già annotato. La copertura si registra per carta, relazione e piano, ma se un piano pertinente rimane incompleto la componente è disattivata nell'intero evento, per tutte le carte e tutti i profili aggiornati esposti. Non mantenere attivi bonus o penalità soltanto per i piani conosciuti meglio. Lo stesso principio vale per altre componenti confrontabili che avrebbero vantaggi asimmetrici dalla copertura. Le capacità minime non disattivabili restano soggette al requisito del 100%; il loro difetto impedisce la disponibilità della nuova versione.

Un uso parziale è ammesso solo in una modalità diagnostica esplicita, separata dai profili offerti per allenamento: il manifest elenca i piani coinvolti e quei risultati non possono sostenere la promozione. Fixture speculare: due piani con pari qualità, sostegno, disponibilità e altri contributi, diversi soltanto per lo stato di annotazione, non cambiano preferenza quando la componente incompleta è disattivata. La correzione è completare i dati, non compensare la lacuna alterando i voti o i pesi per colore.

La conoscenza statica ammessa comprende catalogo dell'evento, ricetta e distribuzioni pubbliche supportate, posti e regole di passaggio. Consente stime approssimate di sostituibilità e supporti futuri; non consente di leggere le buste già generate ma non osservate. Le incertezze restano esplicite.

### Costruzione, match e criteri di confronto

Per la nuova valutazione, gli eventi Draft ordinari assumono BO3 come default strategico dichiarato, modificabile in BO1. È un parametro del valutatore da introdurre, non una proprietà già implementata del tavolo o una descrizione di un torneo reale. Un contesto non riconosciuto usa soltanto main e opzioni di costruzione, senza bonus di sideboard, segnalando la lacuna. Il valore di sideboard BO3 resta contenuto e subordinato a main, mana e completamento, con matchup ipotetici del formato e mai pool avversari letti di nascosto.

Il costruttore mira normalmente alla dimensione minima del main prevista dall'evento. In Limited ordinario 40 è il minimo legale, non il massimo: [CR 100.2b, 100.4b e 100.5](https://media.wizards.com/2026/downloads/MagicCompRules%2020260925.txt), versione del 25 settembre 2026 verificata il 29 ottobre. Eventuali liste più lunghe richiedono una ragione dichiarata. Il numero di carte draftate resta derivato dalle buste e dall'evento. Le carte con impieghi alternativi restano nella lista una sola volta, mantenendo scenari alternativi di uso coerenti; si evita il doppio credito pieno, senza decidere anticipatamente che saranno sempre ciclate. Per la distinzione fra sottotipo e supertipo nel landcycling vale CR 702.29e dello stesso documento: cercare una Mountain non richiede che sia base.

Le basi aggiungibili secondo l'evento sono risorse del costruttore, non pick necessarie. Per valutare uno splash si confrontano configurazioni candidate che possono includerle, pagando slot di main e perdita di consistenza dei colori principali. Il fixing viene valutato insieme alle terre di ciascuna configurazione, senza richiedere che il main corrente abbia già accolto lo splash. La fixture senza basi di splash controlla una configurazione che non le include, non la loro assenza dal pool. Non si presumono aggiungibili terre che l'evento richiede di possedere nel pool.

Protocollo BOT-13: fissare prima della taratura di ciascuna capacità versione dei dati, costruttore diagnostico, rubriche e split di semi/casi. La parte riservata è usata alla valutazione del candidato, senza tararvi la stessa versione; un ulteriore candidato richiede nuovi casi riservati o dichiara il riuso come limite. La revisione del campione di pick e mazzi nasconde l'identità della politica e applica la stessa rubrica: legalità e disponibilità delle carte, main costruibile, mana e turni utili, curva funzionale, risposte, piano di vittoria e affidabilità delle sinergie. Pubblicare i motivi dei disaccordi, senza trasformare un parere in verità universale. Le fixture congelate coprono tutti i criteri BOT, comprese situazioni speculari, ordine dei dati e copie dello stesso nome con sinergie reali.

Il benchmark usa una sola versione della politica per sessione. Tavolo misto significa profili diversi della stessa versione; tavolo omogeneo significa stesso profilo ai posti dei bot. La v3 e la nuova versione si confrontano su sessioni distinte appaiate per configurazioni e buste iniziali, senza introdurre versioni miste nello stesso record. Verificare l'identità delle buste per posto e giro con impronta registrata: lo stesso seme da solo non basta se cambia lo schema casuale. Conservare o riprodurre la collazione originaria con dati e seme fissati; la nuova separazione riguarda il consumo delle decisioni dei bot. Le buste future rimangono accessibili soltanto al banco di prova, mai al valutatore dei bot. Nei confronti Adattivo/specialisti della nuova versione ruotare i posti con flussi isolati; le divergenze successive delle buste passate sono effetti legittimi delle scelte.

Il costruttore diagnostico usa la stessa versione per tutti i pool e dichiara se condivide codice o funzione obiettivo con il valutatore candidato. In quel caso le sue metriche misurano coerenza interna, non costituiscono una prova indipendente di forza. La promozione richiede anche esito delle aspettative strategiche congelate e revisione cieca dei mazzi secondo la rubrica; non può dipendere solo dalla metrica che il candidato ottimizza. Non è obbligatorio sviluppare un secondo costruttore per ottenere questa distinzione.

Promozione: tutti i controlli funzionali e le fixture strategiche concordate passano; nessuna regressione critica irrisolta di legalità, informazioni nascoste, mana impossibile o compatibilità; nel campione riservato il miglioramento sui difetti bersaglio è motivato e non compensato da deterioramenti materiali negli altri assi della rubrica. In caso di esito ambiguo il candidato resta sperimentale, non si inventa una soglia di WR per promuoverlo. Registrare tempo di valutazione e limiti della ricerca su ambiente dichiarato: la ricerca deve restare utilizzabile nel tavolo browser e non è richiesta un'ottimizzazione esaustiva.

Ordine di lavoro: isolare versione e replay storici, banco di prova e casualità per posto; completare dati minimi e costruttore con mana, applicandolo prima ai pool v3 come baseline; integrare sinergie e valore marginale; completare segnali, pivot, curva e counterpick; valutare il profilo Adattivo e la sua eventuale promozione. Le spiegazioni crescono a ogni passo. Il confronto non arriva soltanto a fine lavoro.

### Proposta condizionata di profilo: Adattivo competitivo

È proposto un profilo aggiuntivo come configurazione adattiva senza preferenza stilistica fissa del motore condiviso corretto. Il suo nome non attesta forza competitiva. Il valore principale sta nelle capacità comuni, non nell'aggiunta di un decimo vettore di pesi o di un motore separato.

Nella nuova versione tutti i profili, tranne Potenza pura, usano costruttore, mana e valore marginale condivisi. Gli specialisti mantengono preferenze esplicite su piano, rischio e condizioni di cambio; Archetipo forzato conserva il vincolo scelto. Potenza pura rimane il controllo sul valore individuale. Le politiche v3 restano disponibili per replay e confronto storico. I nuovi profili sono selezionabili soltanto rispettando Dati e disponibilità. Confrontare Adattivo sia con la v3 sia con gli specialisti aggiornati distingue il beneficio delle capacità comuni da quello dell'adattamento.

Adattivo è la configurazione di riferimento senza preferenze stilistiche aggiuntive. Lettura dei segnali aggiornata privilegia esplicitamente la confidenza negli indizi ricevuti e mantiene più a lungo opzioni di piano rispetto alla configurazione di riferimento, entro i vincoli del mazzo. Almeno una fixture motivata deve mostrarne una divergenza utile su stessi pool e busta. Se la valutazione riservata non rileva differenze materiali nella scelta e nel piano, i due vengono unificati in un solo profilo esposto. Non si impongono errori o rumore artificiale per giustificare un decimo nome. La scelta di aggiungere Adattivo separatamente dipende quindi dal confronto; gli identificatori storici restano risolvibili per i replay.

Stato richiesto: pool e lista provvisoria, opzioni di colori e piano, fonti utilizzabili, densità delle sinergie, bisogni scoperti e memoria dei segnali con confidenza. Osservazioni ammesse: proprie pick e buste ricevute, numero di pick, posti e verso di passaggio, oltre alla conoscenza statica dell'evento sopra definita. Azione: scelta della carta e aggiornamento del piano; negli eventi con più carte per turno valutare anche la combinazione nella stessa busta. Transizione: aggiornata dalle osservazioni, con casualità esplicita e riproducibile solo dove giustificata. Politica: confrontare contributo al mazzo, opzioni ancora utili e costo di cambio, senza conoscere contenuti futuri o pool avversari.

Il confronto finale include la baseline Lettura dei segnali v3 su sessioni distinte e appaiate, e Adattivo contro gli specialisti aggiornati su tavoli misti e omogenei della nuova versione, secondo il protocollo. Ruotare profili fra posti e controllare l'ordine dei dati. Usare casi di sinergia, aggro, controllo e splash realmente coperti dai dati. La promozione a predefinito richiede miglioramenti motivati sui casi e sui mazzi esaminati, con regressioni e limiti dichiarati. La creazione separata del profilo e la sua promozione restano sviluppi condizionati successivi a questa analisi.

## Consultazione completa per formato di gioco

Quando si chiedono tutte le carte di un'espansione ammesse in Pauper, o in un altro formato, serve l'intero catalogo pertinente filtrato per legalità, comprese le carte senza recensioni o risultati. Espansione selezionata, formato Constructed ed evento Limited sono dimensioni distinte.

Distinguere legalità alla data dello snapshot, ammissibilità prevista all'uscita e valutazione competitiva. Nelle anteprime `not_legal` non prova l'esclusione futura: una previsione richiede fonte, data e basi verificabili, altrimenti è incerta. Dichiarare cataloghi parziali. Per Pauper contano eventuali stampe comuni precedenti ammesse e banlist, non solo la rarità della stampa corrente. Ospiti, ristampe e prodotti collegati mantengono condizioni proprie.

I limiti delle versioni precedenti sono documentati nei [resoconti storici](history/2026-10-22.md). Per lo stato corrente dei filtri e degli indici condivisi consultare [statistiche](statistics.md) e verificare i JSON e le route dell'espansione interessata. La legalità futura richiede comunque una verifica separata.

Nelle risposte sulla consultazione indicare percorso e link filtrato quando realmente supportato. Se manca la vista, dirlo; un filtro di rarità non è un sostituto del filtro di legalità.

## Analisi completa e consultabilità

Una richiesta di analisi per ogni carta nel contesto del sito comprende consultazione dell'intero catalogo, ricerca, filtri, dettagli per formato, prezzi e fonti. La selezione editoriale è una vista aggiuntiva; l'appendice Commander resta separata dal Limited.

Ricerca, filtri e dettagli espandibili sono la base funzionale. Gli ordinamenti per adozione documentata seguono il contratto delle statistiche. Confronti affiancati e ordinamenti per variazione di prezzo o solidità delle prove sono possibili sviluppi successivi, non requisiti aggiunti automaticamente né funzionalità già disponibili. Un giudizio sulla UX deve verificare quanto è facile trovare, leggere e confrontare le informazioni.

## Indici e metagame per ogni espansione

La base comune comprende WR Draft, WR Sealed e prospettive Standard, Pioneer, Modern e Pauper, con filtri e ordinamenti secondo il [contratto delle statistiche](statistics.md). Per le prospettive, le liste di torneo offrono una misura documentabile dell'adozione e segnali da seguire. Recensioni strategiche e previsioni restano contenuti distinti.

Mostrare le lacune e distinguere assenza osservata, campione assente, carta non legale e dato non applicabile; non assegnare punteggi per rendere completo un ordinamento.
