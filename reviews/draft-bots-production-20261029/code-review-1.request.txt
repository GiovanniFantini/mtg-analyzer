The minimum profile is a functional classification, not a complete Oracle
interpreter. Unsupported contextual relations disable the whole component.
"""
import hashlib

from pathlib import Path

VERSION = 1
COLORS = 'WUBRG'
TYPES = ('Plains', 'Island', 'Swamp', 'Mountain', 'Forest')


def describe(card):
    text = re.sub(r'\([^)]*\)', '', card.get('oracle', ''))
    symbols = re.findall(r'\{([^}]+)\}', card.get('cost', ''))
    roles = ['land' if card.get('land') else 'creature' if card.get('creature') else 'spell']
    def stat(value):
        return int(str(value)) if str(value).isdigit() else None
    power, toughness = stat(card.get('power')), stat(card.get('toughness'))
    for role, pattern in {
        'interaction': r'destroy target|exile target|counter target|target creature gets -|deals? (?:\d+|X) damage to (?:any )?target|deals damage equal to|fights? (?:up to one )?(?:another )?target creature|return target (?:creature|nonland|permanent).*owner.s hand',
        'draw': r'draw (?:a|two|three|\d+) cards?',
        'defense': r'\bdefender\b|prevent.*damage',
        'evasion': r'\bflying\b|can.t be blocked|\bmenace\b|\btrample\b',
        'ramp': r'search your library.*land.*battlefield',
        'sideboard': r'destroy target (?:artifact|enchantment)|exile.*graveyard',
    }.items():
        if re.search(pattern, text, re.I):
            roles.append(role)
    abilities = []
    for line in text.splitlines():
        if ':' not in line:
            continue
        cost, effect = line.split(':', 1)
        if not re.search(r'\badd\b', effect, re.I):
            continue
        produced = list(COLORS) if re.search(r'mana of any color', effect, re.I) else sorted(set(re.findall(r'\{([WUBRGC])\}', effect)))
        if produced:

def validate_config(config, ids, plans=None):
    if not isinstance(config, dict) or config.get('schemaVersion') != 1:
        raise ValueError('botPolicy: schemaVersion non supportata')
    if config.get('match', 'BO3') not in ('BO1', 'BO3', 'unknown'):
        raise ValueError('botPolicy: contesto match non supportato')
    if not isinstance(config.get('cards', {}), dict) or not set(config.get('cards', {})) <= set(ids):
        raise ValueError('botPolicy: carta sconosciuta')
    relations = config.get('relations', [])
    if not isinstance(relations, list) or any(not isinstance(r, dict) for r in relations) or len({r.get('id') for r in relations}) != len(relations):
        raise ValueError('botPolicy: relazioni duplicate o non valide')
    for relation in relations:
        if not isinstance(relation.get('id'), str) or not relation['id']:
            raise ValueError('botPolicy: identità relazione mancante')
        if any(field not in relation for field in ('referenceSupport', 'saturation', 'strength')):
            raise ValueError('botPolicy: riferimento della relazione mancante')
        for field in ('referenceSupport', 'referencePayoffs', 'saturation', 'strength', 'threshold', 'resourceBudget'):
            value = relation.get(field, 0)
            if type(value) not in (int, float) or not 0 <= value <= 100:
                raise ValueError('botPolicy: parametro relazione non valido')
        if relation.get('saturation', 8) < relation.get('referenceSupport', 0):
            raise ValueError('botPolicy: saturazione inferiore al riferimento')
    plan_coverage = config.get('planCoverage', {})
    if not isinstance(plan_coverage, dict) or plans is not None and not set(plan_coverage) <= set(plans):
        raise ValueError('botPolicy: piano sconosciuto')
    for state in plan_coverage.values():
        if not isinstance(state, dict) or state.get('status') not in ('supported', 'not_applicable', 'unsupported') or not isinstance(state.get('provenance'), dict) or not isinstance(state.get('relations', []), list) or not set(state.get('relations', [])) <= {r['id'] for r in relations}:
            raise ValueError('botPolicy: copertura del piano non valida')
    for profile in config.get('cards', {}).values():
        if not isinstance(profile, dict) or not isinstance(profile.get('provenance'), dict):
            raise ValueError('botPolicy: provenienza profilo mancante')
        if profile.get('status') not in ('supported', 'unsupported', 'not_applicable'):
            raise ValueError('botPolicy: stato profilo non valido')
        if 'roles' in profile and (not isinstance(profile['roles'], list) or any(not isinstance(x, str) for x in profile['roles'])):
            raise ValueError('botPolicy: ruoli non validi')
        if 'symbols' in profile and (not isinstance(profile['symbols'], list) or any(not isinstance(s, str) or not re.fullmatch(r'[0-9WUBRGCXYP/]+', s) for s in profile['symbols'])):
            raise ValueError('botPolicy: simboli non validi')
        if 'variable' in profile and type(profile['variable']) is not bool:
            raise ValueError('botPolicy: costo variabile non valido')
        for key in ('mana', 'curve', 'reference', 'synergy'):
            if key in profile and not isinstance(profile[key], dict):
                raise ValueError('botPolicy: descrittore non valido: ' + key)
        reference = profile.get('reference', {})
        if reference.get('status') == 'supported' and not isinstance(reference.get('context'), str):
            raise ValueError('botPolicy: contesto di riferimento mancante')
        for key in ('support', 'payoffs'):
            values = reference.get(key, {})
            if not isinstance(values, dict) or not set(values) <= {r['id'] for r in relations} or any(type(v) not in (int, float) or not 0 <= v <= 100 for v in values.values()):
                raise ValueError('botPolicy: riferimento per carta non valido')
        traits = profile.get('synergy', {})
        if traits and traits.get('status') not in ('supported', 'not_applicable', 'unsupported'):
            raise ValueError('botPolicy: stato sinergia non valido')
        mana = profile.get('mana', {})
        for ability in mana.get('abilities', []):
            if not isinstance(ability, dict) or not isinstance(ability.get('colors'), list) or any(c not in COLORS + 'C' for c in ability['colors']) or type(ability.get('restricted')) is not bool or not isinstance(ability.get('activation'), str):
                raise ValueError('botPolicy: abilità di mana non valida')
        for cycle in mana.get('cycling', []):
            if cycle.get('kind') not in ('basic', 'subtype') or cycle.get('type') not in ('Basic Land', *TYPES) or not isinstance(cycle.get('cost'), str):
                raise ValueError('botPolicy: landcycling non valido')
        for axis in ('enablers', 'payoffs'):
            values = profile.get('synergy', {}).get(axis, {})
            if not isinstance(values, dict) or not set(values) <= {r['id'] for r in relations} or any(type(v) not in (int, float) or not 0 <= v <= 100 for v in values.values()):
                raise ValueError('botPolicy: supporti della relazione non validi')


def payload(cards, active, event, config, table_ready, plans=None):
    profiles = {}
    for card in cards:
        if card['id'] not in active:
            continue
        profile = describe(card)
        override = config.get('cards', {}).get(card['id'], {})
        for key, value in override.items():
            if key in ('mana', 'curve', 'synergy', 'reference'):
                profile[key] = {**profile[key], **value}
            else:
                profile[key] = value
        profiles[card['id']] = profile
    active_cards = [c for c in cards if c['id'] in active]
    minimum = bool(active_cards) and all(c.get('basic') or (
        type(c.get('score')) in (int, float) and 0 <= c['score'] <= 5
        and profiles[c['id']]['status'] == 'supported' and profiles[c['id']].get('roles')
        and isinstance(profiles[c['id']].get('mana'), dict)) for c in active_cards)
    relations = config.get('relations', [])
    plan_coverage = config.get('planCoverage', {})
    covered = bool(relations) and bool(plans) and all(plan_coverage.get(plan, {}).get('status') in ('supported', 'not_applicable') for plan in plans) and all(
        p.get('synergy', {}).get('status') in ('supported', 'not_applicable')
        and p.get('reference', {}).get('status') == 'supported' for p in profiles.values())
    semantic = {'version': VERSION, 'profiles': {
        key: {k: ({key: v[key] for key in ('status', 'support', 'payoffs') if key in v} if k == 'reference' else v) for k, v in value.items() if k != 'provenance'} for key, value in profiles.items()},
        'relations': [{k: r[k] for k in ('id', 'referenceSupport', 'referencePayoffs', 'saturation', 'strength', 'threshold', 'resource', 'resourceBudget') if k in r} for r in relations], 'match': config.get('match', 'BO3'),
        'plans': {key: {'status': value['status'], 'relations': value.get('relations', [])} for key, value in plan_coverage.items()},
        'minimum': event['minDeckSize'], 'extraBasics': event.get('extraBasicLands', False),
        'maxCopies': event.get('maxCopies')}
    revision = hashlib.sha256(json.dumps(semantic, sort_keys=True, ensure_ascii=True).encode()).hexdigest()
    missing = []
    if not table_ready:
        missing.append('Tavolo non disponibile per questo evento.')
    if not minimum:
        missing.append('Voti o profili minimi incompleti sulle stampe attive.')

    gaps.append('Costi ridotti, ruoli e fonti condizionate usano approssimazioni dichiarate, non un simulatore di partita.')

    algorithm_revision = hashlib.sha256(b'\0'.join((root / name).read_bytes() for name in ('draft-policy.js', 'draft-engine.js', 'collation.js'))).hexdigest()
    return {'schemaVersion': 1, 'inferenceVersion': VERSION, 'revision': revision, 'algorithmRevision': algorithm_revision,
            'available': bool(table_ready and minimum), 'status': 'experimental',
            'missing': missing, 'gaps': gaps, 'coverage': {
                'active': len(active_cards), 'minimum': sum(bool(p.get('roles')) and p['status'] == 'supported' for p in profiles.values()),
                'synergies': covered,'plans': {plan: plan_coverage.get(plan, {'status':'unsupported'}) for plan in plans or []}}, 'profiles': profiles, 'relations': relations,
            'match': config.get('match', 'BO3'), 'minDeckSize': event['minDeckSize'],
            'extraBasicLands': event.get('extraBasicLands', False), 'maxCopies': event.get('maxCopies'),
            'allowedBasics': list(COLORS) if event.get('extraBasicLands') else [],
            'defaultEngineVersion': 3, 'exposedAdaptive': False}

def read_json(path):
    def invalid(value):raise ValueError('Numero JSON non valido: '+value)
    return json.loads(path.read_text(encoding='utf-8-sig'),parse_constant=invalid)


def confined(base, relative):
    path = (base / relative).resolve()
    if not path.is_relative_to(base.resolve()):
        raise ValueError(f'Percorso esterno alla cartella: {relative}')
    return path


def normalize_card(source, document):
    if not isinstance(source,dict):raise ValueError(document+': carta non valida')
    card=dict(source)
    if document=='printings.json':card.setdefault('url','')
    identity=str(card.get('id','senza id'))
    def fail(field):raise ValueError(f'{document}/{identity}: campo carta mancante o non valido: {field}')
    for field in ('id','name','set','type','rarity','image','url'):
        if not isinstance(card.get(field),str):fail(field)
    if not card['id'] or not card['name']:fail('id / name')
    if type(card.get('number')) not in (int,str):fail('number')
    for field in ('colors','identity'):
        if not isinstance(card.get(field),list) or any(c not in list('WUBRG') for c in card[field]):fail(field)
    for canonical,alias,types in (('cost','mana',(str,)),('mv','cmc',(int,float)),('oracle','text',(str,))):
        value=card.get(canonical,card.get(alias))
        if type(value) not in types:fail(canonical+' / '+alias)
        card[canonical]=value;card[alias]=value
    card.setdefault('full',card['name']);card.setdefault('large',card['image'])
    card.setdefault('faces',[]);card.setdefault('keywords',[]);card.setdefault('prints',[]);card.setdefault('legalities',{})
    main_type=card['type'].split(' // ')[0]
    for field,default in (('basic',bool(re.search(r'\bBasic\b',main_type) and re.search(r'\bLand\b',main_type))),('land',bool(re.search(r'\bLand\b',main_type))),('creature',bool(re.search(r'\bCreature\b',main_type)))):
        card.setdefault(field,default)
        if type(card[field]) is not bool:fail(field)
    if not isinstance(card['full'],str):fail('full')
    if not isinstance(card['faces'],list) or any(not isinstance(f,dict) or not isinstance(f.get('name'),str) for f in card['faces']):fail('faces')
    card['faces']=[{'mana_cost':'','oracle_text':'','type_line':'','power':'','toughness':'',**face} for face in card['faces']]
    if not isinstance(card['keywords'],list) or any(not isinstance(k,str) for k in card['keywords']):fail('keywords')
    if not isinstance(card['prints'],list) or not isinstance(card['legalities'],dict):fail('prints / legalities')
    if any(not isinstance(p,dict) or not isinstance(p.get('finishes',[]),list) for p in card['prints']):fail('prints.finishes')
    card['prints']=[{'number':'','image':'','finishes':[],**printing} for printing in card['prints']]
    return card


class Expansion:
    def __init__(self, manifest_path):
        self.path = Path(manifest_path)
        self.directory = self.path.parent
        self.manifest = read_json(self.path)
        m = self.manifest
        if m.get('schemaVersion') != 1:
            raise ValueError(f'{self.path}: schemaVersion non supportata')
        if not re.fullmatch(r'[a-z0-9-]+', m.get('id','')):
            raise ValueError('Identificatore espansione non valido')
        for field in ('name','code','date','description','datasets','presentation','storage','features'):
            if not m.get(field): raise ValueError(f'{m["id"]}: manca {field}')
        for field in ('name','code','date','description'):
            if not isinstance(m[field],str):raise ValueError(f'{m["id"]}: {field} deve essere una stringa')
        self.data = {key: read_json(confined(self.directory, m['datasets'][key]))
                     for key in DATASETS if key in m['datasets']}
        missing = set(DATASETS) - self.data.keys()
        if missing: raise ValueError(f'{m["id"]}: documenti mancanti: {sorted(missing)}')
        for key in ('products','limited'):
            if self.data[key].get('schemaVersion')!=1:raise ValueError(f'{m["id"]}: versione {key} non supportata')
        for key in ('cards','printings'):
            if not isinstance(self.data[key].get('cards'),list):raise ValueError(f'{m["id"]}/{key}.json: manca cards')
            self.data[key]['cards']=[normalize_card(card,key+'.json') for card in self.data[key]['cards']]
        if type(self.data['cards'].get('complete',False)) is not bool:raise ValueError(m['id']+'/cards.json: complete deve essere booleano')
        self.data['cards'].setdefault('print_count',len(self.data['printings']['cards']) or sum(max(1,len(card['prints'])) for card in self.data['cards']['cards']))
        self.validate()
        self.availability = self.capabilities()

    @property
    def cards(self):
        ratings = self.data['ratings'].get('cards',{})
        sentiments = self.data['sentiment'].get('cards',{})
        result=[]
        for source in self.data['cards']['cards']:
            card={'score':None,'role':'','why':'','condition':'','evidence':[],'community':None,
                  'forum_note':'','forum_link':'','build':False,**source, **ratings.get(source['id'],{})}
            if source['id'] in sentiments: card['sentiment']=sentiments[source['id']]
            card['keywords']=list(source.get('keywords',[]))
            for alias in self.data['ui'].get('keywordAliases',[]):
                if re.search(alias['pattern'],card['oracle'],re.I) and alias['keyword'] not in card['keywords']:card['keywords'].append(alias['keyword'])
            found=[combo for combo in self.data['combos'] if card['id'] in combo['cards']]
            families={}
            for combo in found:
                q=.65*combo['practical']+.35*combo['power']
                families[combo['family']]=max(q,families.get(combo['family'],0))
            scores=sorted(families.values(),reverse=True)
            score=min(5,scores[0]+min(.5,.2*sum(max(0,x-2) for x in scores[1:4]))) if scores else None
            card.update(combos=[c['id'] for c in found],comboFamilies=len(families),comboScore=int(score*2+.5)/2 if score is not None else None)
            result.append(card)
        return result

    def default_event_id(self, kind):
        """Prefer a usable event of this kind, keeping historical IDs when possible."""

    def event_availability(self,event_id):
        event=self.data['limited']['events'].get(event_id)
        product=self.product_for(event_id)
        recipe=product.get('collation') or {}
        ids=self.product_printings(event_id)
        catalog=self.printing_catalog()
        missing=[]
        kind=event['kind'] if event else 'draft'
        if not event:missing.append(REASONS['limited:'+kind])
        if not self.data['cards'].get('complete'):missing.append(REASONS['completeCards'])
        if not recipe.get('verified'):missing.extend([REASONS['collation:sealed' if kind=='sealed' else 'collation'],*product.get('missing',[])])
        if not ids or any(not catalog.get(id_) for id_ in ids):missing.append(REASONS['catalogMapping:'+kind])
        missing.extend('Stampa senza carta del catalogo: '+id_+'.' for id_ in sorted(ids) if not catalog.get(id_))
        if kind=='draft':
            if not ids or any(not catalog.get(id_) or not (catalog[id_].get('basic') or catalog[id_].get('score') is not None) for id_ in ids):missing.append(REASONS['ratingsComplete'])
            if not self.data['archetypes']:missing.append(REASONS['archetypes'])
        return {'available':not missing,'missing':list(dict.fromkeys(missing))}

    def draft_event_availability(self,event_id):

        return self.event_availability(event_id)

    def draft_format(self, event_id='draft'):
        event_id=self.resolve_event_id(event_id)
        event = self.data['limited']['events'][event_id]
        product = self.data['products']['products'][event['product']]
        archetypes = self.data['archetypes']
        catalog = self.printing_catalog()
        cards=[]
        for printing in self.data['printings']['cards']:
            card = {**printing}
            review = catalog.get(card['id']) or {}
            card.update(score=review.get('score'),role=review.get('role',''))
            card.update(power=review.get('power'),toughness=review.get('toughness'),faces=review.get('faces',[]))
            card['catalogId'] = review.get('id')
            card['archetypes'] = [k for k,a in archetypes.items() if card['name'] in a['core']+a['support']]
            cards.append(card)
        active=self.product_printings(event_id)
        engine_fields=('id','full','colors','identity','cost','mv','rarity','basic','land','creature','score','role','oracle','archetypes')
        spec=product.get('collation') or {}
        engine={'engineVersion':2,'event':{key:event.get(key) for key in ('id','seats','packsPerPlayer','picksPerTurn','directions')},
                'packSize':product['packSize'],
                'collation':{key:spec.get(key) for key in ('schemaVersion','groups','partners','recipe')},
                'cards':[{key:card.get(key) for key in engine_fields} for card in cards if card['id'] in active],
                'archetypeOrder':list(archetypes)}
        revision = hashlib.sha256(json.dumps(engine,sort_keys=True,ensure_ascii=True).encode()).hexdigest()
        compatibility=self.manifest.get('draftReplayCompatibility',{}).get(event_id,{})
        previous=compatibility.get('previousRevisions',[]) if compatibility.get('revision')==revision else []
        from bot_profiles import payload
        bot_policy=payload(cards, active, event, event.get('botPolicy', {}),
                           event.get('kind')=='draft' and self.draft_event_availability(event_id)['available'], archetypes)
        return {'version':1,'id':event['id'],'set':self.manifest['code'],'name':event['name'],
                'botPolicy':bot_policy,'minDeckSize':event['minDeckSize'],
                'extraBasicLands':event.get('extraBasicLands',False),'maxCopies':event.get('maxCopies'),
                'seats':event['seats'],'rounds':event['packsPerPlayer'],'packSize':product['packSize'],'productName':product['name'],
                'picksPerTurn':event['picksPerTurn'],'directions':event.get('directions',[]),
                'assumptions':product.get('assumptions',[]),'cards':cards,'archetypes':archetypes,
                'collation':product.get('collation'),'revision':revision,
                'compatibleRevisions':previous,
                'legacyReplay':event.get('legacyReplay',False) and self.manifest.get('legacyDraftRevision') in previous, 'ui':self.data['ui']}

    def runtime(self):
        return {'schemaVersion':1,'id':self.manifest['id'],'name':self.manifest['name'],'code':self.manifest['code'],
                'storage':self.storage,'events':self.data['limited']['events'],
                'routes':{key:p['url'] for key,p in self.manifest['presentation']['pages'].items()},
                'availability':self.availability,'ui':self.data['ui'],
                'defaultEvents':{kind:self.default_event_id(kind) for kind in ('draft','sealed')},
                'eventAvailability':{key:self.event_availability(key) for key in self.data['limited']['events']}}


def load_expansions():
    expansions = [Expansion(confined(ROOT,entry['manifest'])) for entry in read_json(ROOT/'expansions.json')]
    if len({e.manifest['id'] for e in expansions}) != len(expansions): raise ValueError('Espansioni duplicate')
    storage={}
    for expansion in expansions:
        spec=expansion.storage
        keys=[v for k,v in spec.items() if k!='poolPrefix']
        if 'poolPrefix' in spec:
            keys.extend(spec['poolPrefix']+mode+'-v1' for mode in expansion.data['limited']['events'])
        if 'draftTable' in spec:
            keys.extend(spec['draftTable']+'-'+mode for mode,event in expansion.data['limited']['events'].items() if event['kind']=='draft' and mode!='draft')
        for key in keys:
            if key in storage and storage[key]!=expansion.manifest['id']:raise ValueError(f'Chiave di salvataggio condivisa da espansioni diverse: {key}')
            storage[key]=expansion.manifest['id']
    return expansions

class BotProfilesTests(unittest.TestCase):
    def test_mana_and_alternative_uses_are_structured(self):
        for oracle in ('Deal 3 damage to any target.', 'This spell deals 3 damage to any target.',
                       'This spell deals X damage to target creature.',
                       "Target creature you control fights target creature you don't control.",
                       'Destroy target creature.', 'Exile target creature.'):
            self.assertIn('interaction', describe({'cost': '{R}', 'oracle': oracle})['roles'], oracle)
        artifact = describe({'cost': '{2}', 'type': 'Artifact Creature', 'oracle': ''})
        self.assertTrue(artifact['curve']['artifact'])
        p = describe({'oracle': 'Flying\nMountaincycling {2}', 'cost': '{5}{U}', 'mv': 6, 'creature': True})
        self.assertEqual(p['mana']['cycling'], [{'kind': 'subtype', 'type': 'Mountain', 'cost': '{2}'}])
        self.assertIn('evasion', p['roles'])
        self.assertEqual(p['reference']['status'], 'unsupported')
        restricted = describe({'oracle': '{T}: Add {R}. Spend this mana only to cast creature spells.', 'cost': '{G}'})
        self.assertTrue(restricted['mana']['abilities'][0]['restricted'])
        modal = describe({'oracle': '', 'cost': '{2}{W}', 'mv': 3, 'creature': True,
                          'faces': [{'type_line': 'Creature'}, {'type_line': 'Land', 'oracle_text': '{T}: Add {W}.'}]})
        self.assertEqual(modal['mana']['alternativeLand']['colors'], ['W'])
        fetch = describe({'land': True, 'cost': '', 'oracle': '{T}, Sacrifice this land: Search your library for a basic land card, put it onto the battlefield tapped, then shuffle.'})
        self.assertEqual(fetch['mana']['fetch']['kind'], 'basic')
        self.assertIn('fixing', fetch['roles'])
        self.assertEqual(describe({'cost': '{3}{G} // {1}{G}', 'mv': 4, 'oracle': ''})['status'], 'unsupported')

    def test_policy_coverage_is_uniform_and_does_not_change_legacy_revision(self):
        e = load_expansions()[0]
        fmt = e.draft_format()
        self.assertTrue(fmt['botPolicy']['available'])
        self.assertFalse(fmt['botPolicy']['coverage']['synergies'])
        old_revision = fmt['revision']
        # Images, source text and annotation provenance are nonfunctional.
        e.data['printings']['cards'][0]['image'] = 'https://example.com/new.png'
        self.assertEqual(e.draft_format()['botPolicy']['revision'], fmt['botPolicy']['revision'])
        e.data['limited']['events']['draft']['botPolicy'] = {'schemaVersion': 1, 'match': 'BO1'}
        updated = e.draft_format()
        self.assertEqual(updated['revision'], old_revision)
        self.assertNotEqual(updated['botPolicy']['revision'], fmt['botPolicy']['revision'])

    def test_invalid_descriptors_rejected(self):
        with self.assertRaisesRegex(ValueError, 'carta sconosciuta'):
            validate_config({'schemaVersion': 1, 'cards': {'unknown': {}}}, [])
        with self.assertRaisesRegex(ValueError, 'parametro'):
            validate_config({'schemaVersion': 1, 'relations': [{'id': 'life', 'referenceSupport': 3, 'saturation': 8, 'strength': float('nan')}]}, [])

    def test_incomplete_minimum_blocks_all_new_profiles(self):
        event = {'minDeckSize': 40, 'extraBasicLands': True}
        cards = [{'id': 'a', 'cost': '{W}', 'mv': 1, 'oracle': '', 'score': None}]
        # Artifact status is functional for reductions, unlike provenance.
        original = [{'id': 'a', 'cost': '{2}', 'mv': 2, 'oracle': '', 'score': 3, 'type': 'Creature'}]
        revised = copy.deepcopy(original)
        revised[0]['type'] = 'Artifact Creature'
        self.assertNotEqual(payload(original, {'a'}, event, {}, True)['revision'],
                            payload(revised, {'a'}, event, {}, True)['revision'])
        self.assertFalse(payload(cards, {'a'}, event, {}, True)['available'])
        self.assertFalse(payload([{**cards[0], 'score': 3}], {'a'}, event, {}, False)['available'])

    def test_partial_relations_disable_both_directions(self):
        event = {'minDeckSize': 40, 'extraBasicLands': True}
        cards = [{'id': x, 'cost': '{W}', 'mv': 1, 'oracle': '', 'score': 3} for x in ('a', 'b')]
        configured = {'schemaVersion': 1, 'relations': [{'id': 'life', 'referenceSupport': 3, 'strength': .6, 'saturation': 8}],
                      'cards': {'a': {'synergy': {'status': 'supported', 'enablers': {}, 'payoffs': {'life': 1}},
                                      'reference': {'status': 'supported', 'context': 'ordinary support'}}}}
        value = payload(cards, {'a', 'b'}, event, configured, True)
        self.assertTrue(value['available'])
        self.assertFalse(value['coverage']['synergies'])

    unittest.main()

### Dati e disponibilità: decisioni progettuali

La qualità di base usa `ratings.json`, con scala e provenienza Limited del progetto. GIH e altre misure osservate restano evidenze separate per rivedere manualmente i voti discordanti, considerando contesto e campione; nessuna sostituzione automatica o media nascosta. Un voto mancante non diventa 2,5.

Prima di applicare scarti contestuali, documentare quale situazione assume il voto della scala usata e le eventuali eccezioni per carta: per esempio carta nel suo mazzo tipico con sostegno ordinario, oppure valore autonomo. Non attribuire retroattivamente un significato uniforme ai voti esistenti se le fonti non lo sostengono. I contributi sono scarti, anche negativi, da quel riferimento, non bonus per benefici già compresi nel voto. Se il riferimento manca è una lacuna da risolvere o disattivare secondo le soglie, non una definizione da inventare. Tre fixture controllano sostegno di riferimento, assente ed eccezionale, senza cambiare il voto originale per farle passare.

Prerequisiti per rendere disponibile la nuova versione: tavolo realmente disponibile e, sul 100% delle stampe attive non base, voto valido e profilo minimo esplicito di costo/mana e ruoli. Le soglie valgono per tutti i profili della nuova versione che usano capacità condivise, non soltanto Adattivo. Per ogni capacità aggiuntiva, tutte le carte attive devono avere uno stato esplicito: descrizione supportata, non pertinente verificato oppure non supportato. Un semplice campo vuoto non vale come non pertinente. Le annotazioni e le inferenze Oracle hanno provenienza, revisione e controlli mirati.

Una capacità è dichiarata coperta solo se nessuna carta pertinente resta non supportata e le relative fixture passano. Se mancano i dati minimi, la nuova versione non è disponibile nell'evento: restano soltanto le politiche storiche già disponibili, senza abilitare nuovi tavoli. Se i dati minimi sono completi ma una capacità aggiuntiva è parziale, tutti i profili aggiornati sono sperimentali con lo stesso elenco di lacune e gli stessi contributi non sostenuti disattivati, con valutazione conservativa e spiegata. Potenza pura e Archetipo forzato non aggirano soglie relative a componenti che usano. Nessuna etichetta di competenza generale finché una meccanica del piano valutato è ignorata. Non occorre codificare ogni possibile interazione: il manifest delle capacità dichiara quelle modellate e il benchmark circoscrive le conclusioni.

La granularità delle capacità è fissata prima di annotazione e taratura. In particolare la componente di sinergie contestuali comprende l'insieme delle relazioni previste per i piani dell'evento, non soltanto il singolo asse già annotato. La copertura si registra per carta, relazione e piano, ma se un piano pertinente rimane incompleto la componente è disattivata nell'intero evento, per tutte le carte e tutti i profili aggiornati esposti. Non mantenere attivi bonus o penalità soltanto per i piani conosciuti meglio. Lo stesso principio vale per altre componenti confrontabili che avrebbero vantaggi asimmetrici dalla copertura. Le capacità minime non disattivabili restano soggette al requisito del 100%; il loro difetto impedisce la disponibilità della nuova versione.

Un uso parziale è ammesso solo in una modalità diagnostica esplicita, separata dai profili offerti per allenamento: il manifest elenca i piani coinvolti e quei risultati non possono sostenere la promozione. Fixture speculare: due piani con pari qualità, sostegno, disponibilità e altri contributi, diversi soltanto per lo stato di annotazione, non cambiano preferenza quando la componente incompleta è disattivata. La correzione è completare i dati, non compensare la lacuna alterando i voti o i pesi per colore.

La conoscenza statica ammessa comprende catalogo dell'evento, ricetta e distribuzioni pubbliche supportate, posti e regole di passaggio. Consente stime approssimate di sostituibilità e supporti futuri; non consente di leggere le buste già generate ma non osservate. Le incertezze restano esplicite.

### Costruzione, match e criteri di confronto

Per la nuova valutazione, gli eventi Draft ordinari assumono BO3 come default strategico dichiarato, modificabile in BO1. È un parametro del valutatore da introdurre, non una proprietà già implementata del tavolo o una descrizione di un torneo reale. Un contesto non riconosciuto usa soltanto main e opzioni di costruzione, senza bonus di sideboard, segnalando la lacuna. Il valore di sideboard BO3 resta contenuto e subordinato a main, mana e completamento, con matchup ipotetici del formato e mai pool avversari letti di nascosto.

Il costruttore mira normalmente alla dimensione minima del main prevista dall'evento. In Limited ordinario 40 è il minimo legale, non il massimo: [CR 100.2b, 100.4b e 100.5](https://media.wizards.com/2026/downloads/MagicCompRules%2020260925.txt), versione del 25 settembre 2026 verificata il 29 ottobre. Eventuali liste più lunghe richiedono una ragione dichiarata. Il numero di carte draftate resta derivato dalle buste e dall'evento. Le carte con impieghi alternativi restano nella lista una sola volta, mantenendo scenari alternativi di uso coerenti; si evita il doppio credito pieno, senza decidere anticipatamente che saranno sempre ciclate. Per la distinzione fra sottotipo e supertipo nel landcycling vale CR 702.29e dello stesso documento: cercare una Mountain non richiede che sia base.

Le basi aggiungibili secondo l'evento sono risorse del costruttore, non pick necessarie. Per valutare uno splash si confrontano configurazioni candidate che possono includerle, pagando slot di main e perdita di consistenza dei colori principali. Il fixing viene valutato insieme alle terre di ciascuna configurazione, senza richiedere che il main corrente abbia già accolto lo splash. La fixture senza basi di splash controlla una configurazione che non le include, non la loro assenza dal pool. Non si presumono aggiungibili terre che l'evento richiede di possedere nel pool.

Protocollo BOT-13: fissare prima della taratura di ciascuna capacità versione dei dati, costruttore diagnostico, rubriche e split di semi/casi. La parte riservata è usata alla valutazione del candidato, senza tararvi la stessa versione; un ulteriore candidato richiede nuovi casi riservati o dichiara il riuso come limite. La revisione del campione di pick e mazzi nasconde l'identità della politica e applica la stessa rubrica: legalità e disponibilità delle carte, main costruibile, mana e turni utili, curva funzionale, risposte, piano di vittoria e affidabilità delle sinergie. Pubblicare i motivi dei disaccordi, senza trasformare un parere in verità universale. Le fixture congelate coprono tutti i criteri BOT, comprese situazioni speculari, ordine dei dati e copie dello stesso nome con sinergie reali.

Il benchmark usa una sola versione della politica per sessione. Tavolo misto significa profili diversi della stessa versione; tavolo omogeneo significa stesso profilo ai posti dei bot. La v3 e la nuova versione si confrontano su sessioni distinte appaiate per configurazioni e buste iniziali, senza introdurre versioni miste nello stesso record. Verificare l'identità delle buste per posto e giro con impronta registrata: lo stesso seme da solo non basta se cambia lo schema casuale. Conservare o riprodurre la collazione originaria con dati e seme fissati; la nuova separazione riguarda il consumo delle decisioni dei bot. Le buste future rimangono accessibili soltanto al banco di prova, mai al valutatore dei bot. Nei confronti Adattivo/specialisti della nuova versione ruotare i posti con flussi isolati; le divergenze successive delle buste passate sono effetti legittimi delle scelte.

Il costruttore diagnostico usa la stessa versione per tutti i pool e dichiara se condivide codice o funzione obiettivo con il valutatore candidato. In quel caso le sue metriche misurano coerenza interna, non costituiscono una prova indipendente di forza. La promozione richiede anche esito delle aspettative strategiche congelate e revisione cieca dei mazzi secondo la rubrica; non può dipendere solo dalla metrica che il candidato ottimizza. Non è obbligatorio sviluppare un secondo costruttore per ottenere questa distinzione.

Promozione: tutti i controlli funzionali e le fixture strategiche concordate passano; nessuna regressione critica irrisolta di legalità, informazioni nascoste, mana impossibile o compatibilità; nel campione riservato il miglioramento sui difetti bersaglio è motivato e non compensato da deterioramenti materiali negli altri assi della rubrica. In caso di esito ambiguo il candidato resta sperimentale, non si inventa una soglia di WR per promuoverlo. Registrare tempo di valutazione e limiti della ricerca su ambiente dichiarato: la ricerca deve restare utilizzabile nel tavolo browser e non è richiesta un'ottimizzazione esaustiva.

Ordine di lavoro: isolare versione e replay storici, banco di prova e casualità per posto; completare dati minimi e costruttore con mana, applicandolo prima ai pool v3 come baseline; integrare sinergie e valore marginale; completare segnali, pivot, curva e counterpick; valutare il profilo Adattivo e la sua eventuale promozione. Le spiegazioni crescono a ogni passo. Il confronto non arriva soltanto a fine lavoro.

### Proposta condizionata di profilo: Adattivo competitivo

È proposto un profilo aggiuntivo come configurazione adattiva senza preferenza stilistica fissa del motore condiviso corretto. Il suo nome non attesta forza competitiva. Il valore principale sta nelle capacità comuni, non nell'aggiunta di un decimo vettore di pesi o di un motore separato.

Nella nuova versione tutti i profili, tranne Potenza pura, usano costruttore, mana e valore marginale condivisi. Gli specialisti mantengono preferenze esplicite su piano, rischio e condizioni di cambio; Archetipo forzato conserva il vincolo scelto. Potenza pura rimane il controllo sul valore individuale. Le politiche v3 restano disponibili per replay e confronto storico. I nuovi profili sono selezionabili soltanto rispettando Dati e disponibilità. Confrontare Adattivo sia con la v3 sia con gli specialisti aggiornati distingue il beneficio delle capacità comuni da quello dell'adattamento.

Adattivo è la configurazione di riferimento senza preferenze stilistiche aggiuntive. Lettura dei segnali aggiornata privilegia esplicitamente la confidenza negli indizi ricevuti e mantiene più a lungo opzioni di piano rispetto alla configurazione di riferimento, entro i vincoli del mazzo. Almeno una fixture motivata deve mostrarne una divergenza utile su stessi pool e busta. Se la valutazione riservata non rileva differenze materiali nella scelta e nel piano, i due vengono unificati in un solo profilo esposto. Non si impongono errori o rumore artificiale per giustificare un decimo nome. La scelta di aggiungere Adattivo separatamente dipende quindi dal confronto; gli identificatori storici restano risolvibili per i replay.

Stato richiesto: pool e lista provvisoria, opzioni di colori e piano, fonti utilizzabili, densità delle sinergie, bisogni scoperti e memoria dei segnali con confidenza. Osservazioni ammesse: proprie pick e buste ricevute, numero di pick, posti e verso di passaggio, oltre alla conoscenza statica dell'evento sopra definita. Azione: scelta della carta e aggiornamento del piano; negli eventi con più carte per turno valutare anche la combinazione nella stessa busta. Transizione: aggiornata dalle osservazioni, con casualità esplicita e riproducibile solo dove giustificata. Politica: confrontare contributo al mazzo, opzioni ancora utili e costo di cambio, senza conoscere contenuti futuri o pool avversari.

Il confronto finale include la baseline Lettura dei segnali v3 su sessioni distinte e appaiate, e Adattivo contro gli specialisti aggiornati su tavoli misti e omogenei della nuova versione, secondo il protocollo. Ruotare profili fra posti e controllare l'ordine dei dati. Usare casi di sinergia, aggro, controllo e splash realmente coperti dai dati. La promozione a predefinito richiede miglioramenti motivati sui casi e sui mazzi esaminati, con regressioni e limiti dichiarati. La creazione separata del profilo e la sua promozione restano sviluppi condizionati successivi a questa analisi.

# Motore Draft e qualità dei bot

## Confini del motore

`collation.js` esegue la ricetta del prodotto per Draft e Sealed. `draft-engine.js` gestisce stato, osservazioni, transizioni e politica dei bot senza DOM, rete o salvataggi; `draft-table.js` gestisce l'interfaccia. `build_draft_engine.py` è un adattatore del renderer comune. Gli input attivi sono i documenti dell'espansione, in particolare `products.json`, `limited.json`, `printings.json`, `ratings.json` e `archetypes.json`.

Posti, buste, carte per scelta e versi di passaggio derivano dall'evento. Slot, rarità, coppie, sostituzioni e finiture derivano dal prodotto. Otto posti, tre buste e 14 carte sono un'applicazione specifica, non il contratto universale. Per collazione e dati mancanti vedere [architecture.md](architecture.md).

## Requisiti dei profili

Prima di iniziare, l'utente deve poter assegnare un profilo distinto a ciascun bot. Le strategie hanno priorità, condizioni di cambio piano e costi di rinuncia effettivamente diversi:

- Potenza pura: privilegia il valore individuale della carta, accettando il rischio di un pool incoerente.
- Bomba e costruzione attorno: usa una carta forte come incentivo per colori e supporti, con condizioni esplicite per cambiare piano.
- Lettura dei segnali: accumula indizi dalle buste ricevute e consolida i colori quando il sostegno è sufficiente.
- Archetipo forzato: segue la combinazione scelta prima di iniziare, anche quando è contesa.
- Sinergie: valorizza enabler e payoff in funzione dei supporti già acquisiti e di quelli ancora necessari.
- Aggro e tempo: cerca giocate iniziali, pressione e interazione coerenti con la curva.
- Controllo e vantaggio: cerca risposte, difesa, risorse e condizioni di vittoria sostenibili.
- Fixing e splash: valuta le fonti e i simboli richiesti prima di aggiungere colori.
- Counterpick: confronta il beneficio dedotto di sottrarre una carta con il costo di rinunciare a una propria giocabile; non conosce i pool avversari.

## Stato, informazione e riproducibilità

Le pick precedenti influenzano colori, piano, curva, mana e supporti. Ogni bot conosce soltanto buste e informazioni osservate. Segnali e counterpick sono deduzioni; non accedono a pool avversari o buste future. Il counterpick considera il costo di rinunciare a una propria giocabile.

Un modello markoviano deve esplicitare stato, osservazioni, azioni, transizioni e politica, conservando la memoria necessaria. Le euristiche attuali non sono un modello addestrato o validato sulle partite e non stimano percentuali di vittoria. Le scelte devono essere riproducibili e spiegabili nella revisione finale con esempi di carte dell'espansione.

La versione dell'algoritmo è distinta dall'impronta funzionale dei dati. Conservare la politica originaria dei record compatibili anche dopo ripresa ed esportazione. Non invalidare replay per sole immagini, fonti o testi. Un record incompatibile deve essere salvato integralmente prima di scrivere quello nuovo; se il backup fallisce non sovrascrivere l'originale.

## Indagine e controlli

Conservare i seguenti casi di regressione quando si modifica la politica, senza trasformarne i vecchi pesi in vincoli universali:

- A sostegno equivalente, l'ordine degli archetipi nei dati non assegna una preferenza iniziale a un colore o coppia.
- La compatibilità delle magie con costi ibridi usa i simboli effettivamente pagabili; le creature con X non valgono automaticamente come normali giocate iniziali in base al solo valore di mana.
- I supporti di sinergia distinti non vengono moltiplicati contando ripetutamente copie equivalenti della stessa carta.
- Le asserzioni sui pesi e sui bonus della v3, compreso il bonus nullo per copie ripetute di certi esempi, descrivono quella politica e devono indicarne esplicitamente la versione prima di cambiare il default. Le nuove politiche seguono i [requisiti competitivi](requirements.md#qualità-competitiva-dei-bot-draft): distinguono densità e diversità, possono riconoscere sinergie reali fra copie dello stesso nome e non assegnano credito duplicato senza un'interazione effettiva. Questa è una richiesta per l'evoluzione, non un comportamento già implementato.
- I replay compatibili continuano con la politica registrata, anche dopo ripresa ed esportazione. Versioni supportate e default si verificano in `draft-engine.js`, senza fissare per sempre la versione corrente nelle istruzioni.
- Durante il draft, l'esportazione ordinaria non rivela pool e log degli avversari; il resoconto dei bot viene aggiunto dall'interfaccia soltanto a sessione conclusa. Questa protezione dell'interfaccia non è una garanzia contro l'ispezione del codice o della memoria di un simulatore locale.

`verify-draft-bots.cjs` e `verify-draft-bots-browser.cjs` conservano i controlli pertinenti su politica, replay e flussi. Leggere codice e fixture prima di attribuire loro una copertura specifica.

Per scelte sospette o modifiche alla politica usare [mtg-draft-bots](../.agents/skills/mtg-draft-bots/SKILL.md). Per i comandi di verifica usare [verification.md](verification.md).

`node audit-draft-bots.cjs 200 reality-fracture draft mixed 3` simula tavoli completi; l'ultimo argomento seleziona la versione della politica, per esempio `2` per il confronto storico. Ispezionare la destinazione del rapporto prima di eseguirlo, per preservare acquisizioni precedenti. I parametri e il confronto v2/v3 del 22 ottobre sono nel [resoconto datato](history/2026-10-22.md).

## Politica contestuale v4

La nuova politica è selezionabile come `Bot contestuali (sperimentali)` negli eventi con dati minimi pronti. Il default rimane v3. Non è stata dichiarata una forza competitiva o una WR. Adattivo esiste come configurazione interna di riferimento del benchmark; non aggiunge un decimo profilo nell'interfaccia finché il confronto non dimostra una distinzione utile da Lettura dei segnali aggiornata.

`draft-policy.js` implementa la valutazione pura e limitata per piani, separata dal motore storico. `bot_profiles.py` produce i descrittori offline da costi, tipi, caratteristiche e testo Oracle del catalogo. `expansion_data.py` li collega alle stampe attive e all'evento, senza condizioni sul nome del set e senza modificare i voti. La build richiede ancora soltanto Python e i file del repository; Node serve alle verifiche, non alla build.

### Dati, copertura e annotazioni

Il runtime `format.botPolicy` contiene versione dei descrittori, impronta dei dati, impronta dell'algoritmo, disponibilità, lacune, copertura per piano, profili e relazioni. Le proprietà aggiunte non cambiano il contratto storico di `format.revision` usato dai replay v2/v3.

Il profilo minimo descrive ruoli funzionali, simboli da pagare, fonti riconosciute, cycling tipizzato o base, fetch di terre base, eventuale faccia terra alternativa, curva e caratteristiche stampate. Il riferimento editoriale del voto e le relazioni causali non vengono dedotti da una semplice parola o dall'appartenenza a un archetipo. Simboli non supportati, costi compositi con `//` e carte senza costo normale richiedono dati/supporto espliciti prima di abilitare la nuova politica.

Le annotazioni compatibili e facoltative sono in `limited.events.<evento>.botPolicy`, secondo [bot-policy.schema.json](../schemas/bot-policy.schema.json): `schemaVersion: 1`, `match`, `relations`, `planCoverage` e `cards` indicizzate per stampa. Un override riporta stato e provenienza; i descrittori di mana, curva, sinergia e riferimento si uniscono ai rispettivi campi inferiti. Riferimenti per carta possono specificare `reference.support` e `reference.payoffs` per asse, sostituendo il riferimento comune della relazione senza alterare il voto originale.

Una relazione dichiara identità, supporto di riferimento, soglia, saturazione e forza del contributo; può dichiarare una risorsa consumata e relativo budget. I profili riportano quantità di enabler e payoff per asse. Ogni piano dell'evento ha uno stato esplicito con provenienza e relazioni pertinenti. La componente di sinergie è attiva soltanto con copertura completa dei piani, stati delle carte e riferimenti dei voti; una lacuna spegne sia bonus sia penalità nell'intero evento. Il risultato è riportato nel runtime e nelle spiegazioni, non mascherato da un voto medio.

I descrittori minimi sono una classificazione conservativa di funzioni riconoscibili, non un interprete completo di Oracle. Costi, fonti e meccaniche non modellati restano limiti dichiarati. Le relazioni incomplete di Reality Fracture sono disattivate: non sono state inventate annotazioni per esporre una funzione apparentemente completa.

### Mazzo e decisione

Il costruttore esplora un insieme limitato di colori e configurazioni di splash, sceglie un main provvisorio e identifica le carte escluse. Le basi aggiungibili derivano dall'evento e sono considerate insieme allo splash. La dimensione obiettivo è quella minima dell'evento, non una costante universale. Alla fine completa con basi ammesse un main corto, conservando la diagnosi delle magie mancanti; se non può aggiungere basi usa soltanto carte possedute, con segnalazione dei ripieghi e dei buchi ancora impossibili da colmare. Rispetta il limite di copie quando previsto.

La qualità marginale confronta il mazzo dopo la carta con il migliore piano precedente, usando la stessa prospettiva temporale e lo stesso contributo dei segnali. Le carte del main conservano il valore assoluto: la media del loro colore nel catalogo non viene sottratta. Soltanto i posti scoperti ricevono un valore di riempimento atteso, derivato da qualità e massa delle marginali pubbliche per ruolo e colore, pick restanti e un fattore prudenziale dichiarato di 0,35. A draft concluso questa attesa vale zero. Esclusioni e dipendenze della collazione sono approssimate solo per queste attese, mentre le buste continuano a usare la ricetta originale. Posti futuri da riempire sono conteggi, mai carte che il bot conosce in anticipo.

I contributi di sinergia sono scarti dal sostegno di riferimento, con valore di opzione iniziale limitato dalla disponibilità attesa e dalle pick restanti. Copie e varietà sono distinte; payoff che consumano la stessa risorsa se ne dividono il budget. Gli enabler ricevono valore anche dai beneficiari presenti. Nei dati parziali questo intero contributo resta zero.

Le fonti vengono valutate al turno pertinente: colori per lanciarle, costo di attivazione, rapidità, ingressi tappati e terre effettivamente ricercabili nella configurazione. Landcycler che competono per poche terre si dividono il contributo. Facce terra e cycling conservano un solo slot e un contributo frazionario alternativo alla minaccia; non contano come due carte. Fonti condizionate non riconosciute non diventano fonti universali. I costi ibridi usano insiemi di fonti pagabili senza contare due volte una terra doppia.

Le stime di accesso al mana usano code ipergeometriche con interpolazione per fonti indirette frazionarie e vincoli di pagamento approssimati separatamente. Gli splash possono usare due o tre basi aggiungibili anche senza fixing posseduto; il loro turno utile dichiarato è due turni dopo il costo nominale. Tutti i colori fuori piano pertinenti sono candidati. Uno slot solo fixing privilegia le fonti dei colori che deve sbloccare; il confronto marginale conserva sempre la possibilità di mantenere la configurazione precedente. Non sono inclusi mulligan o simulazione completa dei turni; le stime non sono WR. Le riduzioni di costo riconosciute dipendono da risorse plausibili del proprio main e non creano automaticamente una giocata iniziale garantita.

I segnali conservano memoria per verso di passaggio e confrontano qualità e profondità residue con un'attesa pubblica dichiarata. Il ritorno della stessa busta e le scelte multiple nello stesso turno non generano osservazioni indipendenti. Nessun pool avversario o busta futura entra nell'evaluator. Fase e cambi di piano seguono progresso, confidenza e mazzi costruibili, senza il costo affondato delle pick già spese. Un miglioramento di un piano alternativo vicino conserva un valore di opzione anche prima di superare il piano corrente: contributo limitato a 0,35, pesato per distanza fra i piani risultanti e pick restanti, nullo all'ultima pick. Il piano già realizzato non riceve nuovamente questo credito. Sono pesi euristici, non probabilità calibrate.

Aggro privilegia pressione e giocate iniziali, Controllo difesa e risposte. Bomba mantiene un incentivo soltanto se l'ancora resta nel main. Segnali dà più peso agli indizi e mantiene più a lungo piani alternativi; Fixing accetta configurazioni aggiuntive solo col mana verificato. Potenza pura rimane il controllo individuale. Counterpick non prevale su un miglioramento proprio o una riserva utile di sideboard; il beneficio dedotto rimane limitato e incerto. BO1 non attribuisce valore al sideboarding; BO3 è l'assunzione predefinita dichiarata del nuovo valutatore.

Pick-Two usa una ricerca limitata sulle combinazioni della stessa busta e registra la motivazione congiunta. Il modello non promette l'ottimo combinatorio. Il sorteggio è ristretto a copie della stessa stampa con uguale funzione nel piano: un punteggio vicino non rende intercambiabili carte diverse. La modalità API `deterministic: true` elimina quel sorteggio.

### Replay e spiegazioni

L'esportazione in corso contiene soltanto il record del giocatore e non i pool/log dei bot.

# Implementazione dei requisiti Draft BOT, 29 ottobre 2026

Politica v4 implementata nel motore condiviso e selezionabile come opzione sperimentale. V3 rimane predefinita; i replay storici continuano con la propria politica. Nessun commit, push o modifica alla pipeline.

## Comportamento implementato

- Profili minimi offline per stampa attiva, separati dai voti. La disponibilità richiede copertura completa dei dati minimi e un tavolo realmente disponibile.
- Main provvisorio e finale costruito dalle carte possedute e dalle basi ammesse dall'evento. Refill dopo le esclusioni; le carte scartate non continuano a fornire mana, curva o supporti.
- Confronto dei piani sul valore assoluto del main, con bisogni funzionali e mana. Solo gli slot ancora vuoti ricevono una stima di riempimento futuro, basata su dati pubblici, ruolo, colore e pick restanti.
- Splash con basi aggiungibili, landcycling tipizzato o base, fetch, facce terra e produttori accessibili. Gli usi alternativi occupano un solo slot; la carta non fornisce il mana per lanciare se stessa. Le stime sono approssimate e dichiarate.
- Relazioni causali con sostegno di riferimento, rendimenti, opzioni e risorse contese, subordinate alla copertura completa di carte e piani.
- Memoria separata per verso di passaggio, qualità/profondità attesa e wheel. Nessun accesso ai pool avversari o alle buste future. Nessun costo affondato per le pick spese.
- Preferenze dei profili, curve aggro e controllo, riserve BO3 e costo opportunità della negazione. BO1 e contesto sconosciuto non ricevono valore di sideboarding.
- Flussi casuali separati per posto e decisione nella nuova versione; collazione storica preservata. Scelte congiunte limitate per Pick-Two e modalità deterministica tramite API.
- Revisione finale con main, fonti, alternative, migliore stimata, sorteggio, busta, pool precedente, sostituzioni e lacune. Anteprime mouse, tastiera e tocco; log avversari disponibili soltanto a draft concluso.
- Record v4 con opzioni e impronte di dati, algoritmo e schema casuale. Backup integrale dei record incompatibili prima della sostituzione; se fallisce, l'originale resta intatto.

## Review indipendente

Richieste, hash degli input, risposte grezze e JSON normalizzati sono conservati in questa cartella. Le review sono eseguite tramite Claude CLI, con modello effettivo `claude-opus-5-5`, senza strumenti e senza persistenza della sessione. Non sono risposte simulate da Codex.

Il primo round ha individuato nove difetti confermati: refill, doppi simboli del colore principale, riduzioni, terre solo incolori, fetch, auto-supporto della carta, costi compositi, impronta della collazione e fase in Pick-Two. Il secondo ne ha individuati sette: obiettivo assoluto dei piani, delta con segnali coerenti, rimozioni rosse/verdi, valore del riempitivo finale, splash con basi, ordine dei candidati e tipo artefatto funzionale. Il terzo ne ha individuati quattro: costo distinto dal turno di splash, modalità solo fixing anche in colore, facce terra con restrizioni e main finale scelto senza segnali. Le risoluzioni sono in [code-decisions.json](code-decisions.json); le regressioni riproducibili sono in [verify-draft-policy-regressions.cjs](../../verify-draft-policy-regressions.cjs) e [test_bot_profiles.py](../../test_bot_profiles.py).

Il quarto round ha aggiunto valore delle opzioni di piano, fetch con basi possedute, esclusione delle terre tappate dominate e spiegazioni separate dei pivot. Il quinto ha rilevato il doppio credito di un piano già diventato migliore: il termine di opzione ora lo esclude. Il sesto ha corretto i falsi pivot dovuti alla sola sideboard, il campo pivot nelle scelte congiunte e il costo opportunità del corpo usato come fixing. Le fixture speculari e le risposte originarie restano conservate. In totale 28 rilievi confermati corretti prima della valutazione riservata.

Implementazione e verifiche funzionali concluse. Il settimo ricontrollo mirato di Opus ha restituito zero rilievi; gli hash dei sorgenti esaminati coincidono con quelli finali. Motore storico, collazione, caricatore e schema non erano cambiati dalle review complete precedenti.

### Valutazione cieca

Opus ha ricevuto 15 mazzi anonimizzati e 15 pick, in due richieste indipendenti senza persistenza. Il giudice delle pick non ha visto le liste finali; nessuno dei due ha ricevuto la mappa fra campione e politica. Catalogo, costi, statistiche stampate e Oracle erano forniti. Le pick sono tutte B2 P7, con pool precedente e busta, senza sequenza completa delle buste precedenti o profilo stilistico.

[Giudizio sui mazzi](blind-review-decks-1.review.json), [giudizio sulle pick](blind-review-picks-1.review.json), [decisioni di lettura](blind-adjudication.json). Entrambe le risposte dichiarano insufficiente l'evidenza per la promozione.

Nelle pick la preferenza di Opus coincide con 2/5 scelte v3 e 6/10 scelte v4; per la v4 ci sono inoltre tre preferenze diverse e un caso incerto. Sono accordi con un giudizio, non percentuali di accuratezza. Per esempio considera coerente Skilled Battlecarver in RG per la curva; preferisce Sphinx's Approach a una seconda Yargle nel campione omogeneo v4, pur riconoscendo il bisogno di creature. Le due prese di terre criticate più nettamente appartengono alla v3; la lettura del secondo caso è troppo categorica, perché il pool contiene già Cruel Calculations blu. Non vengono trasformate in un tasso di errore certo.

Sui mazzi il limite ricorrente è il mana: colori principali poco serviti, splash discutibili e curva talvolta alta. Per esempio il campione `sample-54d07d96` ha dieci carte che richiedono R e cinque fonti rosse fisse; `sample-c35e8337` presenta solo cinque Forest con molte carte G. Sono criticità del costruttore diagnostico condiviso, quindi non possono essere attribuite automaticamente alla sola politica che ha draftato il pool. Fonti indirette e costi ridotti vanno comunque considerati; non tutte le critiche ai duplicati leggendari implicano un taglio corretto.

La valutazione riservata non è stata usata per ritoccare pesi o scelte del candidato. Un'ulteriore taratura di mana, tempo dello splash e gestione delle copie richiederà nuovi casi riservati o una dichiarazione esplicita di riuso. Non è attestata competenza competitiva generale: la v4 resta opt-in sperimentale e Adattivo resta interno al banco di prova.

## Limiti e decisione di disponibilità

Reality Fracture è l'evento Draft realmente disponibile nel catalogo corrente. La copertura minima è 411/411 stampe attive. Gli altri set non vengono abilitati per allargare il benchmark. I profili minimi del set sono inferiti dai dati locali; i voti e la collazione restano quelli originari.

Le annotazioni causali complete e il contesto editoriale dei voti non sono disponibili per tutte le carte e tutti i piani di Reality Fracture. La componente di sinergia resta quindi spenta nell'intero evento, per tutti i profili aggiornati, con lacuna visibile. Le fixture provano il supporto del motore, senza attestare una ricerca per carta già completata.

Le misure del costruttore diagnostico condividono codice e obiettivo con il candidato: indicano coerenza interna, non forza indipendente. Non sono state simulate partite e non viene dichiarata una WR. Il campione limitato di un solo evento e i dati causali mancanti non giustificano una promozione a bot competitivo generale o un cambio del default.
